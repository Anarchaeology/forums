Here is small macro that will extract select area from excel to text file in vboard or EzBoard format. I used this macro in our guild boards to list my inventory, so it will create hyper link to www.auno.org from all textual cells. You can also disable hyper link from the code.
code:
Option Explicit

' Output file ID
Const OUTPUT_FILE As Integer = 1
' How many columns or rows there can be in selection before warning
Const ROW_OR_COLUMN_WARNING_COUNT As Integer = 300

Const ELEMENT_START = "["
Const ELEMENT_END = "]"

' vboard, EzBoard
Const HEADER = ELEMENT_START + "code" + ELEMENT_END
Const FOOTER = ELEMENT_START + "/code" + ELEMENT_END

' -- Item Links --
' Are item links enabled
Const ITEM_LINKS As Boolean = True
' First row that is handled as data row
Const FIRST_DATA_ROW As Integer = 2

' vboard
Const LINK_START_ELEMENT = ELEMENT_START + "url="
Const LINK_END_ELEMENT = ELEMENT_START + "/url" + ELEMENT_END
' EzBoard
'Const LINK_START_ELEMENT = ELEMENT_START + "link="
'Const LINK_END_ELEMENT = ELEMENT_START + "/link" + ELEMENT_END

' Item link.
' $ADDRESS$ is address that allow searching (part of the HTTP link)
' $NAME$ is link's name

' - www.hackersquest.org
'Const INITIAL_LINK As String = LINK_START_ELEMENT + _
    "http://www.hackersquest.org/ao/show.php3?iname=$ADDRESS$" + _
    ELEMENT_END + "$NAME$" + LINK_END_ELEMENT
' - www.auno.org
Const INITIAL_LINK As String = LINK_START_ELEMENT + _
    "http://auno.org/ao/db.php?cmd=search&name=$ADDRESS$" + _
    ELEMENT_END + "$NAME$" + LINK_END_ELEMENT
    
' Delimeter that is used in search phrase to seperate words
Const ADDRESS_WORD_DELIMETER As String = "+"

'----------------------------------------------------------------------------------
' This macro extracts select area to ASCII file.
' Selected area is appended (old data is not deleted) to the file. If file doesn't
' exists already, new file is created.
Sub ExtractToTextFile()
    'Get current range
    Dim rngData As Range
    Set rngData = GetDataRange(ROW_OR_COLUMN_WARNING_COUNT)
    If rngData Is Nothing Then
        Exit Sub
    End If
    
    'Get Filename and open file
    Dim strFileName As String
    strFileName = CreateFileName()
    Close OUTPUT_FILE
    Open strFileName For Append As OUTPUT_FILE
        
    ' Header
    Print #OUTPUT_FILE, HEADER
    
    'Extract Data
    ExtractData OUTPUT_FILE, rngData, ITEM_LINKS, FIRST_DATA_ROW, INITIAL_LINK
    
    ' Footer
    Print #OUTPUT_FILE, FOOTER
    
    ' Close
    Print #OUTPUT_FILE, vbNewLine
    Close OUTPUT_FILE
End Sub
'----------------------------------------------------------------------------------
' Get selected data range that is extract to file.
' If there is more than iRowOrCoulumnWarningCount function gives warning for user.
' User may want to proceed or cancel. If user cancels function returns Nothing.
Function GetDataRange(iRowOrCoulumnWarningCount As Integer) As Range
    Set GetDataRange = Selection
    ' Check row and column count
    If GetDataRange.Rows.Count >= iRowOrCoulumnWarningCount Or _
        GetDataRange.Columns.Count >= iRowOrCoulumnWarningCount Then
        ' Many colums selected. Ask from user is everything ok.
        If MsgBox("Are you sure that you want export selection to file?", _
            vbOKCancel, "Warning: Large area is selected") = vbCancel Then
            Set GetDataRange = Nothing
        End If
    End If
End Function
'----------------------------------------------------------------------------------
' Create file name for text file.
' Filename is constructed from workbook path and name,
' active sheet name and from the current date.
Function CreateFileName() As String
    CreateFileName = ThisWorkbook.Path + "\" + _
        Left(ThisWorkbook.Name, Len(ThisWorkbook.Name) - 4) + _
        " - " + ActiveSheet.Name + _
        " - " + Date$ + ".txt"
End Function
'----------------------------------------------------------------------------------
' Extract rngData to file
' This subprogram will extract selected table to ASCII file. All rows (cells) in
' each column are formated so that their length is same. If needed additional
' spaces are added to each rows in column (cell).
'
' If search links are enabled (bSearchLink is true) each cell with text data in
' selected table in iFirstDataRow and after, are added to the ASCII file as
' a "Search Link". Search link is constructed from the cell's text and from
' strInitialLink (see more info in function CreateLink).
'
Sub ExtractData(iOutputFile As Integer, rngData As Range, bSearchLinks As Boolean, _
    iFirstDataRow As Integer, strInitialLink As String)

    If rngData.Columns.Count <= 0 Then
        Exit Sub
    End If
    
    Dim colData As Range
    Dim iCol As Integer
    ReDim iarrMaxLength(rngData.Columns.Count - 1) As Integer
    
    ' Get maximal length for each column in data
    iCol = 1
    For Each colData In rngData.Columns
        ' First calculate maximal length
        iarrMaxLength(iCol - 1) = GetMaxRowLength(colData)
        iCol = iCol + 1
    Next colData
    
    Dim rowData As Range
    Dim strLine As String
    Dim strCell As String
    Dim iRow As Integer
    
    iRow = 1
    ' Go throuhg each row in data
    For Each rowData In rngData.Rows
        strLine = ""
        iCol = 1
        
        ' Go though each column in row
        For Each colData In rowData.Columns
            strCell = colData.Text
            
            If IsNumeric(colData) Or iRow < iFirstDataRow Or Not bSearchLinks Then
                ' Number don't add hyperlink or
                ' this was header row or
                ' search links are disabled
                strLine = strLine + strCell
            Else
                ' Non number add hyperlink
                strLine = strLine + CreateLink(strInitialLink, strCell, strCell)
            End If
            
            ' Add same spaces after column so that each row in column are same length
            strLine = strLine + Space(iarrMaxLength(iCol - 1) - Len(strCell) + 1)
            
            iCol = iCol + 1
        Next colData
        
        ' Write to file
        Print #iOutputFile, strLine
        
        iRow = iRow + 1
    Next rowData
End Sub
'----------------------------------------------------------------------------------
' Calculates maximal row's length in column and returns it to caller.
Function GetMaxRowLength(colData As Range) As Integer
    Dim rowData As Range
    Dim strRow As String
    
    GetMaxRowLength = 0
    ' Go through each row in column
    For Each rowData In colData.Rows
        strRow = rowData.Text
        If GetMaxRowLength < Len(strRow) Then
            ' Row's length was longer
            GetMaxRowLength = Len(strRow)
        End If
    Next rowData
End Function
'----------------------------------------------------------------------------------
' This function creates new link.
'
' Function creates link's HTTP address from strAddress.
' All multiple spaces are removed from the strAddress and
' spaces are replaced with a ADDRESS_WORD_DELIMETER.
' Finally a processed address is added to the strInitialLink, by replacing
' $ADDRESS$ substring in strInitialLink.
'
' strName is added to the strInitialLink without processing, by replacing
' $NAME$ substring in strInitialLink where address is already added.
'
' Returns: Link to strAddress with strName
Function CreateLink(strInitialLink As String, ByVal strAddress As String, _
    strName As String) As String
    
    Dim lDuplicate As Long
    
    ' Remove all multiple spaces
    lDuplicate = InStr(strAddress, "  ")
    While lDuplicate <> 0
        strAddress = Replace(strAddress, "  ", " ")
        lDuplicate = InStr(strAddress, "  ")
    Wend
    
    ' Replace all spaces with delimeter
    strAddress = Replace(strAddress, " ", ADDRESS_WORD_DELIMETER)
    
    ' Create link
    CreateLink = Replace(strInitialLink, "$ADDRESS$", strAddress)
    CreateLink = Replace(CreateLink, "$NAME$", strName)
End Function

As far as I know crit dmg is:

IF
(MaxDmg-MinDmg)*(1+AMS/400)-AC/10 < 0
THEN
(MinDmg+CritDmg)*(1+AMS/400)
ELSE
(MaxDmg*(1+AMS/400)-AC/10)+CritDmg*(1+AMS/400)
If a mob has 3200 ac (like a backyard guard )and yer using lets say a QL 184 Krutt (29-178 (1143)) with 1000 Skill, you'll crit for 4304 every time on that particular mob.

I think that's what AG's calculator uses.
